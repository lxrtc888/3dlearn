## 🎬 场景介绍模态框

### 功能说明

每个场景加载时，**先显示介绍模态框**，包含：

- 场景标题和图标
- 学习目标
- 操作说明
- 知识点预览
- "开始学习"按钮

用户点击按钮后，模态框关闭，场景**自动开始播放**动画。

## 🎯 核心教学理念

### 设计哲学：让学习变成探索

```
传统教学：看 → 听 → 记
沉浸式教学：探索 → 操作 → 发现 → 理解 → 应用
```

### 五大必须原则

| 原则               | 要求                     | 实现方式                  |
| ------------------ | ------------------------ | ------------------------- |
| **真实可靠** | 物理原理100%正确         | 公式、动画符合科学规律    |
| **强交互**   | 学生主动操作，非被动观看 | 点击、拖拽、滑块、按钮    |
| **教学性**   | 有明确学习目标和步骤     | 引导提示、分步骤教学      |
| **可操作**   | 视角自由控制             | 旋转/缩放/平移 + 重置按钮 |
| **即时反馈** | 操作后立即看到结果       | 动画、数值变化、声音提示  |

### 学习体验金字塔

```
                    ┌─────────────┐
                    │   应用创造   │  ← 学生自己调参数看结果
                   ┌┴─────────────┴┐
                   │    分析理解    │  ← 点击部件看原理解释
                  ┌┴───────────────┴┐
                  │     操作体验     │  ← 拖拽、点击、交互
                 ┌┴─────────────────┴┐
                 │      观察感知      │  ← 3D动画、视觉效果
                └───────────────────┘
```

## 💻 技术实现标准

### 场景类完整结构

```javascript
window.XxxScene = class XxxScene {
    constructor(scene, camera, renderer) {
        this.scene = scene;
        this.camera = camera;
        this.renderer = renderer;
      
        // 核心属性
        this.mainGroup = null;
        this.interactables = [];
        this.highlighted = null;
      
        // 教学属性
        this.currentStep = 0;
        this.teachingSteps = [];
        this.isPlaying = false;
      
        // 默认相机位置（用于重置）
        this.defaultCameraPos = { x: 0, y: 10, z: 30 };
      
        // 场景参数（可调节）
        this.params = {};
    }

    // ========== 必须实现 ==========
  
    init() {
        this.setupCamera();
        this.setupEnvironment();
        this.setupScene();
        this.setupUI();
        this.showInitialGuide();
    }
  
    setupCamera() {
        this.camera.position.set(
            this.defaultCameraPos.x,
            this.defaultCameraPos.y,
            this.defaultCameraPos.z
        );
        this.camera.lookAt(0, 0, 0);
    }
  
    setupEnvironment() {
        // 背景
        this.scene.background = new THREE.Color(0x0a0a15);
        this.scene.fog = new THREE.FogExp2(0x0a0a15, 0.008);
      
        // 地面网格
        const grid = new THREE.GridHelper(60, 60, 0x334466, 0x1a1a2e);
        grid.position.y = -5;
        this.scene.add(grid);
      
        // 光照
        this.setupLights();
    }
  
    setupLights() {
        const ambient = new THREE.AmbientLight(0x404060, 0.6);
        this.scene.add(ambient);
      
        const main = new THREE.DirectionalLight(0xffffff, 1);
        main.position.set(10, 20, 10);
        main.castShadow = true;
        this.scene.add(main);
    }
  
    setupScene() {
        this.mainGroup = new THREE.Group();
        this.mainGroup.position.set(0, 0, 0); // 居中
        this.scene.add(this.mainGroup);
      
        // 创建场景内容...
    }
  
    setupUI() {
        const controlsDiv = document.getElementById('scene-controls');
        controlsDiv.style.display = 'flex';
        controlsDiv.innerHTML = `
            <button class="control-btn" id="btn-play">
                <i class="fas fa-play"></i> 开始演示
            </button>
            <button class="control-btn" id="btn-reset">
                <i class="fas fa-redo"></i> 重新开始
            </button>
            <button class="control-btn" id="btn-reset-view">
                <i class="fas fa-video"></i> 重置视角
            </button>
        `;
      
        this.bindUIEvents();
    }
  
    bindUIEvents() {
        document.getElementById('btn-play').onclick = () => this.togglePlay();
        document.getElementById('btn-reset').onclick = () => this.reset();
        document.getElementById('btn-reset-view').onclick = () => this.resetView();
    }
  
    showInitialGuide() {
        setTimeout(() => {
            this.showGuide('💡 拖动鼠标旋转视角，滚轮缩放');
        }, 1000);
        setTimeout(() => {
            this.showGuide('👆 点击蓝色部件查看详细说明');
        }, 4000);
    }
  
    // ========== 交互方法 ==========
  
    resetView() {
        gsap.to(this.camera.position, {
            x: this.defaultCameraPos.x,
            y: this.defaultCameraPos.y,
            z: this.defaultCameraPos.z,
            duration: 0.8,
            ease: 'power2.out'
        });
        this.camera.lookAt(0, 0, 0);
    }
  
    togglePlay() {
        this.isPlaying = !this.isPlaying;
        const btn = document.getElementById('btn-play');
        btn.innerHTML = this.isPlaying 
            ? '<i class="fas fa-pause"></i> 暂停' 
            : '<i class="fas fa-play"></i> 开始演示';
    }
  
    reset() {
        this.isPlaying = false;
        this.currentStep = 0;
        // 重置场景状态...
    }
  
    highlightObject(target) {
        // 取消之前高亮
        if (this.highlighted?.material?.emissive) {
            this.highlighted.material.emissive.setHex(
                this.highlighted.userData.originalEmissive || 0
            );
        }
      
        // 新高亮
        if (target.material?.emissive) {
            target.userData.originalEmissive = target.material.emissive.getHex();
            target.material.emissive.setHex(0x00ffff);
        }
        this.highlighted = target;
      
        // 缩放动画
        gsap.to(target.scale, { 
            x: 1.1, y: 1.1, z: 1.1, 
            duration: 0.15, 
            yoyo: true, 
            repeat: 1 
        });
    }
  
    showInfoPanel(target) {
        const panel = document.getElementById('info-panel');
        document.getElementById('info-title').innerHTML = 
            `<i class="fas fa-cube mr-2"></i>${target.userData.name}`;
        document.getElementById('info-content').innerHTML = 
            target.userData.description;
        panel.classList.add('visible');
    }
  
    showGuide(message) {
        // 在场景内显示引导提示
        const container = document.getElementById('scene-canvas-container');
        const guide = document.createElement('div');
        guide.className = 'scene-guide-message';
        guide.innerHTML = message;
        container.appendChild(guide);
      
        setTimeout(() => guide.classList.add('visible'), 100);
        setTimeout(() => {
            guide.classList.remove('visible');
            setTimeout(() => guide.remove(), 300);
        }, 3000);
    }
  
    // ========== 生命周期 ==========
  
    animate(time, delta) {
        // 动画逻辑
    }
  
    getInteractables() {
        return this.interactables;
    }
  
    dispose() {
        if (this.mainGroup) this.scene.remove(this.mainGroup);
    }
  
    onBackgroundClick() {
        const panel = document.getElementById('info-panel');
        if (panel) panel.classList.remove('visible');
      
        if (this.highlighted?.material?.emissive) {
            this.highlighted.material.emissive.setHex(
                this.highlighted.userData.originalEmissive || 0
            );
            this.highlighted = null;
        }
    }
};
```
